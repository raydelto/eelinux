C_ARES_VERSION = 1.34.9
C_ARES_SITE = https://github.com/c-ares/c-ares/releases/download/v$(C_ARES_VERSION)
C_ARES_INSTALL_STAGING = YES
C_ARES_LICENSE = MIT
C_ARES_LICENSE_FILES = LICENSE.md
C_ARES_CPE_ID_VENDOR = c-ares
C_ARES_INTEGRATED_TOOLS = YES
C_ARES_CONF_OPTS = \
	-DCARES_BUILD_TOOLS=ON \
	-DCARES_INSTALL=ON \
	-DCARES_BUILD_TESTS=OFF \
	-DCARES_RANDOM_FILE=/dev/urandom \
	-DCARES_THREADS=$(if $(BR2_TOOLCHAIN_HAS_THREADS),ON,OFF)

ifeq ($(BR2_STATIC_LIBS),y)
C_ARES_CONF_OPTS += -DCARES_SHARED=OFF -DCARES_STATIC=ON
else
# Export the matching static SDK archive alongside the shared runtime.
C_ARES_CONF_OPTS += -DCARES_SHARED=ON -DCARES_STATIC=ON
endif

HOST_C_ARES_CONF_OPTS = \
	-DCARES_BUILD_TOOLS=ON -DCARES_INSTALL=ON -DCARES_BUILD_TESTS=OFF \
	-DCARES_RANDOM_FILE=/dev/urandom -DCARES_SHARED=ON -DCARES_STATIC=ON

$(eval $(cmake-package))
$(eval $(host-cmake-package))
